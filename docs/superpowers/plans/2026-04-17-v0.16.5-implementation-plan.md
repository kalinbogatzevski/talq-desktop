> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship v0.16.5 adding Twemoji-based emoji (autoreplace + Telegram-style picker), a proper image viewer with pan/zoom/Esc, and a split welcome screen with changelog plus a way to return to it mid-session.

**Architecture:** One shared emoji data/pixmap layer (`src/core/EmojiData`) feeds the picker widget, composer autoreplace, chat-body rendering, and reaction pills. The image viewer is a standalone top-level window using `QGraphicsView`, with single-instance tracking on `MainWindow`. The welcome screen becomes a horizontal `QSplitter` (server card left, `QTextBrowser` right) reachable via a home signal from either a clickable logo or a 🏠 button in the sidebar header.

**Tech Stack:** Qt 6.8.2 (Widgets, GraphicsView, Network), MinGW 13.1 / MSYS2, CMake+Ninja, Inno Setup for installers. Twemoji v14 PNGs bundled as Qt resource. No test framework beyond the existing integration harness `tests/call_test.cpp` — verification is manual via the dev build (`bash scripts/deploy-dev.sh`).

```python
#!/usr/bin/env python3
"""Generate EmojiData_generated.cpp from emoji-datasource JSON.

Usage: python scripts/gen-emoji-data.py resources/emoji-data-source.json src/core/EmojiData_generated.cpp
"""
import json
import sys
from pathlib import Path

CATEGORY_MAP = {
    "Smileys & Emotion": "Smileys",
    "People & Body":     "People",
    "Animals & Nature":  "Animals",
    "Food & Drink":      "Food",
    "Activities":        "Activities",
    "Travel & Places":   "Travel",
    "Objects":           "Objects",
    "Symbols":           "Symbols",
    "Flags":             "Flags",
}

# Short-form text shortcuts — must be maintained manually since the dataset
# doesn't ship them.
SHORT_FORMS = {
    "1F642": [":)", ":-)"],                 # slightly_smiling_face
    "1F603": [":D", ":-D"],                 # smiley
    "1F61B": [":P", ":-P", ":p", ":-p"],    # stuck_out_tongue
    "1F61E": [":(", ":-("],                 # disappointed_face
    "1F609": [";)", ";-)"],                 # wink
    "1F62E": [":o", ":-o", ":O", ":-O"],    # open_mouth
    "1F610": [":|", ":-|"],                 # neutral_face
    "1F615": [":/", ":-/"],                 # confused
    "2764":  ["<3"],                        # heart
    "1F494": ["</3"],                       # broken_heart
    "1F618": [":*", ":-*"],                 # kissing_heart
    "1F62D": [":'("],                       # loudly_crying_face
    "1F60E": ["B)", "B-)"],                 # sunglasses
}


def unified_to_codepoints(unified: str) -> str:
    """'1F600' -> '\\U0001F600' escape sequence for C++."""
    return "".join(
        f"\\U{int(c, 16):08X}" for c in unified.split("-")
    )


def cxx_string(s: str) -> str:
    """Escape arbitrary Python string for a C++ QStringLiteral argument."""
    escaped = s.replace("\\", "\\\\").replace('"', '\\"')
    return f'QStringLiteral("{escaped}")'


def main(src_json: Path, out_cpp: Path):
    data = json.loads(src_json.read_text(encoding="utf-8"))

    lines = [
        '// GENERATED by scripts/gen-emoji-data.py — DO NOT EDIT.',
        '#include "EmojiData.h"',
        '',
        'namespace EmojiData {',
        '',
        'void loadGeneratedData(QVector<EmojiEntry> &out)',
        '{',
        f'    out.reserve({len(data)});',
        '',
    ]

    for e in data:
        cat = CATEGORY_MAP.get(e.get("category", ""))
        if not cat:
            continue  # skip component emojis (skin tones themselves)
        cp = unified_to_codepoints(e["unified"])
        name = e.get("name") or e.get("short_name", "")
        shortcodes = [f":{sn}:" for sn in e.get("short_names", [])]
        short_forms = SHORT_FORMS.get(e["unified"], [])
        variants = []
        for tone_key, tone_data in (e.get("skin_variations") or {}).items():
            variants.append(unified_to_codepoints(tone_data["unified"]))
        keywords = e.get("short_names", []) + [name.lower()]

        lines.append('    out.append({')
        lines.append(f'        QStringLiteral(u"{cp}"),')
        lines.append(f'        Category::{cat},')
        lines.append(f'        {cxx_string(name.lower())},')
        lines.append('        {' + ", ".join(cxx_string(s) for s in shortcodes) + '},')
        lines.append('        {' + ", ".join(cxx_string(s) for s in short_forms) + '},')
        if variants:
            lines.append('        {' + ", ".join(f'QStringLiteral(u"{v}")' for v in variants) + '},')
        else:
            lines.append('        {},')
        lines.append('        {' + ", ".join(cxx_string(k) for k in keywords) + '},')
        lines.append('    });')

    lines += [
        '}',
        '',
        '} // namespace EmojiData',
        '',
    ]

    out_cpp.write_text("\n".join(lines), encoding="utf-8")
    print(f"Wrote {out_cpp} ({len(data)} entries)")


if __name__ == "__main__":
    main(Path(sys.argv[1]), Path(sys.argv[2]))
```

- [ ] **Step 1: Replace the `pixmapFor` stub with the real implementation**

```cpp
QPixmap pixmapFor(const QString &codepoints, int sizePx)
{
    QString key = codepoints + QLatin1Char(':') + QString::number(sizePx);
    auto it = g_pixmapCache.constFind(key);
    if (it != g_pixmapCache.constEnd())
        return it.value();

    // Build the hex filename used by Twemoji: codepoints joined by '-', lowercased,
    // U+FE0F (variation selector) stripped because Twemoji filenames omit it.
    QString hex;
    for (int i = 0; i < codepoints.size(); ) {
        uint cp = codepoints[i].unicode();
        if (QChar::isHighSurrogate(cp) && i + 1 < codepoints.size()) {
            cp = QChar::surrogateToUcs4(codepoints[i], codepoints[i+1]);
            i += 2;
        } else {
            ++i;
        }
        if (cp == 0xFE0F) continue;  // variation selector — omitted in filenames
        if (!hex.isEmpty()) hex += '-';
        hex += QString::number(cp, 16);
    }

    QString path = QStringLiteral(":/twemoji/72x72/%1.png").arg(hex);
    QPixmap pm(path);
    if (!pm.isNull() && pm.height() != sizePx)
        pm = pm.scaled(sizePx, sizePx, Qt::KeepAspectRatio, Qt::SmoothTransformation);

    g_pixmapCache.insert(key, pm);
    return pm;
}
```

```cpp
QVector<const EmojiEntry*> inCategory(Category c)
{

QVector<const EmojiEntry*> search(const QString &query)
{
    QString q = query.trimmed().toLower();
    if (q.isEmpty()) return {};

    QVector<QPair<int, const EmojiEntry*>> ranked;
    for (const auto &e : g_entries) {
        int score = 0;
        if (e.name.contains(q)) score += (e.name.startsWith(q) ? 10 : 5);
        for (const QString &k : e.keywords)
            if (k.contains(q)) score += (k == q ? 8 : 2);
        for (const QString &sc : e.shortcodes)
            if (sc.contains(q)) score += (sc == ":" + q + ":" ? 15 : 3);
        if (score > 0) ranked.append({score, &e});
    }

    std::sort(ranked.begin(), ranked.end(),
              [](const auto &a, const auto &b) { return a.first > b.first; });

    QVector<const EmojiEntry*> out;
    out.reserve(ranked.size());
    for (const auto &p : ranked) out.append(p.second);
    return out;
}
```

```cpp
void pushRecent(const EmojiEntry *e)
{
    if (!e) return;
    g_recentCodepoints.removeAll(e->codepoints);
```

```cpp
#include "core/EmojiData.h"

void ComposerWidget::handleAutoreplace()
{
    QTextCursor cur = m_input->textCursor();
    int pos = cur.position();
    if (pos < 3) return;  // need at least ":) " or similar

    QString text = m_input->toPlainText();
    // Only trigger on trailing space (or newline).
    if (pos == 0 || !(text[pos - 1] == ' ' || text[pos - 1] == '\n')) return;

    // Look back up to 10 chars for a short-form or :shortcode: before the space.
    int start = qMax(0, pos - 11);
    QString tail = text.mid(start, pos - 1 - start);

    // Try short-form first (2–3 chars, no colons except leading).
    for (int n = qMin(3, tail.size()); n >= 2; --n) {
        QString cand = tail.right(n);
        const auto *e = EmojiData::findByShortForm(cand);
        if (e) {
            QTextCursor c = m_input->textCursor();
            c.beginEditBlock();
            c.setPosition(pos - 1 - n);
            c.setPosition(pos - 1, QTextCursor::KeepAnchor);
            c.insertText(e->codepoints);
            c.endEditBlock();
            return;
        }
    }
}
```

```cpp
static bool isEmojiCluster(const QString &cluster)
{
    // Quick heuristic: any char is Extended Pictographic OR sits in emoji ranges.
    for (int i = 0; i < cluster.size(); ) {
        uint cp = cluster[i].unicode();
        if (QChar::isHighSurrogate(cp) && i + 1 < cluster.size()) {
            cp = QChar::surrogateToUcs4(cluster[i], cluster[i+1]);
            i += 2;
        } else ++i;
        if ((cp >= 0x1F300 && cp <= 0x1FAFF) ||  // Misc + Supplemental Symbols
            (cp >= 0x2600  && cp <= 0x27BF)  ||  // Misc Symbols + Dingbats
            (cp >= 0x1F1E6 && cp <= 0x1F1FF) ||  // Regional Indicators (flags)
            cp == 0x00A9 || cp == 0x00AE)       // © ®
            return true;
    }
    return false;
}
```

```cpp
        QFont f = p->font();

void ChatPainter::paintMessageEmojis(QPainter *p, const MessageLayout &ml, qreal offsetY)
{
    if (ml.emojiRuns.isEmpty() || !ml.bodyDoc) return;

    QAbstractTextDocumentLayout *layout = ml.bodyDoc->documentLayout();
    const qreal bodyLeft = ml.bodyRect.left();
    const qreal bodyTop  = ml.bodyRect.top() + offsetY;

    for (const auto &r : ml.emojiRuns) {
        QTextCursor c(ml.bodyDoc.get());
        c.setPosition(r.docPosition);
        QRectF rect = layout->blockBoundingRect(c.block());
        // Compute x via QTextLine::cursorToX for line-accurate placement.
        QTextBlock block = c.block();
        for (auto it = block.begin(); !it.atEnd(); ++it) {
            QTextFragment frag = it.fragment();
            if (frag.position() <= r.docPosition &&
                frag.position() + frag.length() > r.docPosition) {
                QTextLayout *lay = block.layout();
                if (!lay) break;
                for (int li = 0; li < lay->lineCount(); ++li) {
                    QTextLine line = lay->lineAt(li);
                    int lineStart = line.textStart();
                    int lineEnd   = lineStart + line.textLength();
                    if (r.docPosition >= lineStart && r.docPosition < lineEnd) {
                        qreal x = bodyLeft + line.cursorToX(r.docPosition);
                        qreal y = bodyTop + rect.top() + line.y();
                        int sz = int(r.widthPx * 1.1); // slight upscale
                        drawEmoji(p, r.codepoints, QPointF(x, y), sz);
                        break;
                    }
                }
                break;
            }
        }
    }
}
```

```cpp
void ApiClient::fetchFileImage(int fileId, std::function<void(const QImage &, const QString &)> cb)
{
    // Nextcloud serves raw file at /remote.php/dav/files/<user>/<path>, but
    // we don't have the path here. Use /index.php/core/preview?fileId=…&x=4096&y=4096&a=1
    // for a large preview — works for any image up to ~4096 edge.
    QUrl url(serverUrl());
    url.setPath("/index.php/core/preview");
    QUrlQuery q;
    q.addQueryItem("fileId", QString::number(fileId));
    q.addQueryItem("x", "4096");
    q.addQueryItem("y", "4096");
    q.addQueryItem("a", "1");
    url.setQuery(q);

    QNetworkRequest req(url);
    req.setRawHeader("OCS-APIRequest", "true");
    req.setRawHeader("Authorization", basicAuthHeader());
    auto *reply = m_nam->get(req);
    QObject::connect(reply, &QNetworkReply::finished, [reply, cb]() {
        reply->deleteLater();
        if (reply->error() != QNetworkReply::NoError) { cb(QImage(), reply->errorString()); return; }
        QImage img = QImage::fromData(reply->readAll());
        cb(img, img.isNull() ? QStringLiteral("decode failed") : QString());
    });
}
```
